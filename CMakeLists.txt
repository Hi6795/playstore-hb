cmake_minimum_required(VERSION 3.20)
project(playstore_hb VERSION 0.1.0 LANGUAGES CXX)
option(PLAYSTOREHB_BUILD_DESKTOP_CONTRACT "Build portable C++ platform-contract executable" ON)
option(PLAYSTOREHB_BUILD_PS4 "Build OpenOrbis PS4 client" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(PLAYSTOREHB_BUILD_DESKTOP_CONTRACT)
  add_executable(playstorehb-platform-contract apps/ps4-client/src/application.cpp apps/ps4-client/src/simulator_platform.cpp apps/ps4-client/src/contract_main.cpp)
  target_include_directories(playstorehb-platform-contract PRIVATE apps/ps4-client/include)
  if(MSVC)
    target_compile_options(playstorehb-platform-contract PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(playstorehb-platform-contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
if(PLAYSTOREHB_BUILD_PS4)
  message(FATAL_ERROR "OpenOrbis v0.5.4 has no supported CMake integration. Run scripts/build-ps4.ps1 on Windows or scripts/build-ps4 on POSIX.")
endif()
